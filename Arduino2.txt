int trig[4] = {2, 4, 6, 8};
int echo[4] = {3, 5, 7, 9};

int ledVerde[4]    = {10, 12, A0, A2};
int ledVermelho[4] = {11, 13, A1, A3};

bool vagaOcupada[4];
unsigned long tempoEntrada[4];

float medirDistancia(int i) {
  digitalWrite(trig[i], LOW);
  delayMicroseconds(2);
  digitalWrite(trig[i], HIGH);
  delayMicroseconds(10);
  digitalWrite(trig[i], LOW);
  long duracao = pulseIn(echo[i], HIGH, 30000);
  if (duracao == 0) return 999.0;
  return duracao * 0.034 / 2.0;
}

void setup() {
  Serial.begin(9600);
  for (int i = 0; i < 4; i++) {
    pinMode(trig[i], OUTPUT);
    pinMode(echo[i], INPUT);
    pinMode(ledVerde[i], OUTPUT);
    pinMode(ledVermelho[i], OUTPUT);
    vagaOcupada[i] = false;
    digitalWrite(ledVerde[i], HIGH);
    digitalWrite(ledVermelho[i], LOW);
  }
  Serial.println("V:4");
}

void loop() {
  int vagasLivres = 0;
  for (int i = 0; i < 4; i++) {
    float distancia = medirDistancia(i);
    if (distancia > 0 && distancia < 200.0) {
      if (!vagaOcupada[i]) {
        vagaOcupada[i] = true;
        tempoEntrada[i] = millis();
        digitalWrite(ledVerde[i], LOW);
        digitalWrite(ledVermelho[i], HIGH);
      }
    } else {
      if (vagaOcupada[i]) {
        unsigned long duracaoMs = millis() - tempoEntrada[i];
        float horas = duracaoMs / 3600000.0;
        vagaOcupada[i] = false;
        digitalWrite(ledVermelho[i], LOW);
        digitalWrite(ledVerde[i], HIGH);
        Serial.print("P:");
        Serial.print(i + 1);
        Serial.print(":");
      }
    }
    if (!vagaOcupada[i]) vagasLivres++;
  }
  Serial.print("V:");
  Serial.println(vagasLivres);
  delay(300);
}
